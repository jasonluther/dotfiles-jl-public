# Debian-only apt packages, installed in addition to common.txt on Linux.
# Format: one package name per line. Blank lines and `#` comments allowed.
#
# Bootstrap-time essentials (curl, ca-certificates, gnupg, gh repo)
# are installed by scripts/linux/base.sh — keep those there since
# they need to run before chezmoi apply.

build-essential

# 1Password CLI (`op`). Lives in 1Password's apt repo, not Debian's —
# scripts/linux/setup-1password-cli.sh registers the source list and
# signing key before install-packages.sh runs. The macOS install is
# handled by the `1password-cli` cask in the Brewfile.
1password-cli

# Google Chrome — the self-hosted CI runners' Playwright e2e suites launch
# channel=chrome (real branded Chrome, not bundled Chromium; the only
# supported chromium lane on Ubuntu 26.04). Lives in Google's apt repo —
# scripts/linux/setup-google-chrome.sh registers the source list and
# signing key before install-packages.sh runs (amd64-only; other arches
# get filtered out as no-candidate). macOS installs Chrome out of band.
google-chrome-stable

# Syscall tracer, for diagnosing wedged processes on the self-hosted CI
# runners — a hung browser driver shows a blocked syscall vs a spin, which
# `py-spy` alone can't distinguish. Linux-only by nature: macOS ships dtruss,
# which SIP blocks against most binaries, so there's no common.txt equivalent.
strace

# Login shell — dot_zshrc/dot_zprofile are useless without it. macOS ships
# zsh by default, but Debian/Ubuntu defaults to bash. The chsh-to-zsh
# chezmoi script switches the login shell after install.
zsh

# Node.js comes via NodeSource (scripts/linux/setup-nodesource.sh, run
# before this installer by the `00-nodesource` chezmoi script). NodeSource's
# nodejs package bundles npm, so the apt `npm` package isn't needed.
# `nodejs` itself is listed in common.txt; the NodeSource repo controls
# which major version apt installs.
