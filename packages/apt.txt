# Debian-only apt packages, installed in addition to common.txt on Linux.
# Format: one package name per line. Blank lines and `#` comments allowed.
#
# Bootstrap-time essentials (curl, ca-certificates, openssh-server, gh repo)
# are installed by bootstrap/linux/modules/base.sh — keep those there since
# they need to run before chezmoi apply.

build-essential

# Login shell — dot_zshrc/dot_zprofile are useless without it. macOS ships
# zsh by default, but Debian/Ubuntu defaults to bash. The chsh-to-zsh
# chezmoi script switches the login shell after install.
zsh

# `nodejs` (installed via common.txt + alias `node=nodejs`) ships without npm
# on Debian/Ubuntu — unlike brew's node, which bundles it. Adding npm here
# keeps the npm.txt-driven global installer working on Linux too.
npm
